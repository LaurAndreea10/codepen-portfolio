    function booScary(n) {
  if (!Number.isSafeInteger(n) || n < 1)
    throw new RangeError('Positive integer required');
  if (n % 15 === 0) return 'BooScary';
  if (n % 3 === 0) return 'Boo';
  if (n % 5 === 0) return 'Scary';
  return String(n);
}
